15. GROUP BY YEAR AND QUARTER
==================================================

16. GROUP BY AGE GROUP
==================================================

Query:
SELECT CASE
         WHEN age < 30 THEN 'Under 30'
         WHEN age BETWEEN 30 AND 50 THEN '30-50'
         ELSE 'Over 50'
       END AS age_group,
       COUNT(*) AS employee_count, AVG(salary) AS avg_salary
FROM employees_lab13
GROUP BY CASE
           WHEN age < 30 THEN 'Under 30'
           WHEN age BETWEEN 30 AND 50 THEN '30-50'
           ELSE 'Over 50'
         END
ORDER BY age_group;

21. MULTIPLE AGGREGATIONS
==================================================

Query:
SELECT department, COUNT(*) AS employee_count,
       MIN(salary) AS min_salary, MAX(salary) AS max_salary,
       AVG(salary) AS avg_salary, STDDEV(salary) AS salary_std_dev
FROM employees_lab13
GROUP BY department

22. CONDITIONAL AGGREGATION
==================================================

Query:
SELECT department, COUNT(*) AS total_employees,
       COUNT(CASE WHEN salary > 50000 THEN 1 END) AS high_earners,
       COUNT(CASE WHEN hire_date > '2023-01-01' THEN 1 END) AS recent_hires
FROM employees_lab13
GROUP BY department
ORDER BY department;
