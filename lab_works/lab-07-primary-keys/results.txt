5. SINGLE-COLUMN PRIMARY KEY WITH SERIAL

Command:

CREATE TABLE students (
    student_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE,
    faculty VARCHAR(50)
);

Result:

CREATE TABLE

Inserted 3 rows.

\d students

Table "public.students"

   Column   |          Type          | Collation | Nullable | Default
------------+------------------------+-----------+----------+----------------------------------------------
 student_id | integer                |           | not null | nextval('students_student_id_seq'::regclass)
 name       | character varying(100) |           | not null |
 email      | character varying(150) |           |          |
 faculty    | character varying(50)  |           |          |

Indexes:
    "students_pkey" PRIMARY KEY, btree (student_id)
    "students_email_key" UNIQUE CONSTRAINT, btree (email)

11. PRIMARY KEY CONSTRAINTS

Command:

CREATE TABLE departments_lab7 (
    dept_id INTEGER PRIMARY KEY,
    dep_name VARCHAR(100)
);

Result:

CREATE TABLE

First insert:

INSERT INTO departments_lab7
VALUES
    (1, 'Computer Science');

Result:

INSERT 0 1

Duplicate primary key test:

INSERT INTO departments_lab7
VALUES
    (1, 'Mathematics');

Result:

ERROR: duplicate key value violates unique constraint
"departments_lab7_pkey"

DETAIL:
Key (dept_id)=(1) already exists.

NULL primary key test:

INSERT INTO departments_lab7
VALUES
    (NULL, 'Physics');

Result:

ERROR: null value in column "dept_id" of relation
"departments_lab7" violates not-null constraint

DETAIL:
Failing row contains (null, Physics).

Result: PRIMARY KEY prevents duplicate and NULL values.
